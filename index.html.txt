<script>
  const SCRIPT_URL = "https://script.google.com/a/macros/flipkart.com/s/AKfycbxkRry2tSWZVDGyi4ATycYz9kxy69Lz2AM5wIppFviAAOcccq8aazUjNOAB0zJheR2vpw/exec"; // https://script.google.com/macros/s/.../exec

  const el = id => document.getElementById(id);
  const esc = s => String(s == null ? "" : s).replace(/[&<>"]/g, c => ({ "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;" }[c]));
  const toteVal = () => el("toteId").value.trim().toUpperCase();

  function toast(msg) {
    const t = el("toast");
    t.textContent = msg;
    t.style.display = "block";
    clearTimeout(toast._t);
    toast._t = setTimeout(() => { t.style.display = "none"; }, 2200);
  }

  async function apiCall(params) {
    const url = new URL(SCRIPT_URL);
    Object.keys(params).forEach(k => url.searchParams.append(k, params[k]));
    const res = await fetch(url.toString());
    return await res.json();
  }

  function formPayload() {
    return {
      toteId: toteVal(),
      vendor: el("vendor").value.trim(),
      pickerName: el("picker").value.trim(),
      tlNo: el("tl").value.trim(),
      totalQty: el("totalQty").value.trim(),
      rejectQty: el("rejectQty").value.trim()
    };
  }

  function calcMetrics() {
    const total = Number(el("totalQty").value) || 0;
    const reject = Number(el("rejectQty").value) || 0;
    el("mTotal").textContent = total;
    el("mReject").textContent = reject;
    el("mNet").textContent = Math.max(0, total - reject);
  }

  async function loadTote() {
    const tote = toteVal();
    if (!tote) return toast("Enter Tote ID");
    toast("Checking Tote...");
    try {
      const res = await apiCall({ action: "get", toteId: tote });
      el("heroTote").textContent = tote;
      el("pickCard").classList.remove("hidden");
      el("viewCard").classList.add("hidden");
      el("histCard").classList.add("hidden");

      if (res && res.success && res.data) {
        el("vendor").value = res.data.vendor || "";
        el("picker").value = res.data.pickerName || "";
        el("tl").value = res.data.tlNo || "";
        el("totalQty").value = res.data.totalQty || "";
        el("rejectQty").value = res.data.rejectQty || "0";
        const st = res.data.status || "";
        el("vrpCard").classList.toggle("hidden", !(st.includes("Complete") || st.includes("VRP")));
        toast("Tote Loaded!");
      } else {
        ["vendor","picker","tl","totalQty"].forEach(i => el(i).value = "");
        el("rejectQty").value = "0";
        el("vrpCard").classList.add("hidden");
        toast("New Tote Ready");
        el("vendor").focus();
      }
      calcMetrics();
    } catch (err) { toast("Error: " + (err.message || err)); }
  }

  async function saveDraft() {
    if (!toteVal()) return toast("Enter Tote ID");
    toast("Draft Saving...");
    try {
      const res = await apiCall(Object.assign({ action: "save" }, formPayload()));
      toast(res.success ? "Draft Saved!" : "Failed: " + res.message);
    } catch (e) { toast("Error: " + (e.message || e)); }
  }

  async function completePicking() {
    if (!toteVal()) return toast("Enter Tote ID");
    if (!el("picker").value.trim()) return toast("Picker Name required");
    if (!el("totalQty").value.trim()) return toast("Total Qty required");
    toast("Saving...");
    try {
      const res = await apiCall(Object.assign({ action: "saveAndComplete" }, formPayload()));
      if (!res.success) return toast("Failed: " + res.message);
      el("vrpCard").classList.remove("hidden");
      el("vrp").focus();
      toast("Picking Done! Scan VRP");
    } catch (e) { toast("Error: " + (e.message || e)); }
  }

  async function verifyVRP() {
    const vrp = el("vrp").value.trim();
    if (!vrp) return toast("Enter VRP number");
    toast("Verifying...");
    try {
      const res = await apiCall({ action: "vrpVerify", toteId: toteVal(), vrp: vrp });
      if (!res.success) return toast("Failed: " + res.message);
      toast("✅ VRP Verified!");
      el("vrp").value = "";
      viewToteStatus();
    } catch (e) { toast("Error: " + (e.message || e)); }
  }

  async function viewToteStatus() {
    const tote = toteVal();
    if (!tote) return toast("Enter Tote ID first");
    try {
      const res = await apiCall({ action: "get", toteId: tote });
      if (res && res.success && res.data) {
        el("vTote").textContent = res.data.toteId;
        el("vVend").textContent = res.data.vendor || "---";
        el("vPick").textContent = res.data.pickerName || "---";
        el("vUnits").textContent = `${res.data.totalQty || 0} (Rej: ${res.data.rejectQty || 0})`;
        el("vStatus").textContent = res.data.status || "Saved";
        el("vVRP").textContent = res.data.vrp || "---";
        el("viewCard").classList.remove("hidden");
      } else toast("Tote not found");
    } catch (e) { toast(e.message || e); }
  }

  async function viewHistoryLogs() {
    const tote = toteVal();
    if (!tote) return toast("Enter Tote ID first");
    try {
      const res = await apiCall({ action: "history", toteId: tote });
      const h = (res.data && res.data.history) || [];
      el("histLogs").innerHTML = h.length
        ? h.map(r => `<div class="detail-item"><span class="detail-key">${esc(r.resetTimestamp)}</span><span class="detail-val">${esc(r.pickerName)} | ${esc(r.totalQty)}/${esc(r.rejectQty)} | ${esc(r.vrp)}</span></div>`).join("")
        : "No history";
      el("histCard").classList.remove("hidden");
    } catch (e) { toast(e.message || e); }
  }

  async function resetTote() {
    const tote = toteVal();
    if (!confirm(`Dispatch & reset Tote: ${tote}?`)) return;
    toast("Archiving...");
    try {
      const res = await apiCall({ action: "reset", toteId: tote });
      if (!res.success) return toast("Failed: " + res.message);
      ["pickCard","vrpCard","viewCard"].forEach(i => el(i).classList.add("hidden"));
      el("toteId").value = "";
      el("toteId").focus();
      toast("✅ Archived & Reset!");
    } catch (e) { toast(e.message || e); }
  }

  function closeView() { el("viewCard").classList.add("hidden"); }
  function closeHist() { el("histCard").classList.add("hidden"); }

  el("toteId").addEventListener("keydown", e => { if (e.key === "Enter") { e.preventDefault(); loadTote(); } });
  el("vrp").addEventListener("keydown", e => { if (e.key === "Enter") { e.preventDefault(); verifyVRP(); } });
</script>
